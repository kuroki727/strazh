cmake_minimum_required(VERSION 3.16)
project(strazh VERSION 0.1.0 LANGUAGES CXX)

if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
    message(FATAL_ERROR "Strazh requires Linux")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(NOT CMAKE_BUILD_TYPE AND NOT CMAKE_CONFIGURATION_TYPES)
    set(CMAKE_BUILD_TYPE Release CACHE STRING "Build type" FORCE)
endif()
option(STRAZH_ASAN "Enable AddressSanitizer" OFF)
option(STRAZH_UBSAN "Enable UndefinedBehaviorSanitizer" OFF)
option(STRAZH_WERROR "Treat compiler warnings as errors" OFF)

add_library(strazh_options INTERFACE)
target_compile_definitions(strazh_options INTERFACE _GNU_SOURCE)
target_compile_options(strazh_options INTERFACE
    -Wall -Wextra -Wpedantic -Wconversion -Wsign-conversion -Wshadow -Wformat=2)
if(STRAZH_WERROR)
    target_compile_options(strazh_options INTERFACE -Werror)
endif()
foreach(sanitizer ASAN UBSAN)
    if(STRAZH_${sanitizer})
        if(sanitizer STREQUAL "ASAN")
            set(flag -fsanitize=address)
        else()
            set(flag -fsanitize=undefined)
        endif()
        target_compile_options(strazh_options INTERFACE ${flag} -fno-omit-frame-pointer)
        target_link_options(strazh_options INTERFACE ${flag})
    endif()
endforeach()

add_executable(myinit main.cpp)
add_executable(stage1 stage1.cpp)
add_executable(supervisor supervisor.cpp)
add_executable(svctl svctl.cpp)
foreach(target myinit stage1 supervisor svctl)
    target_link_libraries(${target} PRIVATE strazh_options)
endforeach()

install(TARGETS myinit stage1 supervisor RUNTIME DESTINATION etc/myinit)
install(TARGETS svctl RUNTIME DESTINATION bin)

# Tests are opt-in; default builds need no Python or test dependencies.
option(BUILD_TESTING "Build and register isolated regression tests" OFF)
include(CTest)
if(BUILD_TESTING)
    find_package(Python3 3.8 REQUIRED COMPONENTS Interpreter)
    add_executable(test_service tests/test_service.cpp)
    target_link_libraries(test_service PRIVATE strazh_options)
    find_package(Threads REQUIRED)
    add_executable(test_core tests/core.cpp)
    target_link_libraries(test_core PRIVATE strazh_options)
    add_executable(test_control tests/control.cpp)
    target_link_libraries(test_control PRIVATE strazh_options Threads::Threads)
    add_test(NAME core COMMAND test_core $<TARGET_FILE:test_service>)
    add_test(NAME control COMMAND test_control)
    add_test(NAME host-safety COMMAND ${Python3_EXECUTABLE}
        ${CMAKE_CURRENT_SOURCE_DIR}/tests/integration.py $<TARGET_FILE_DIR:supervisor> --host-only)
    add_test(NAME integration COMMAND ${Python3_EXECUTABLE}
        ${CMAKE_CURRENT_SOURCE_DIR}/tests/integration.py $<TARGET_FILE_DIR:supervisor>)
    set_tests_properties(core control host-safety integration PROPERTIES TIMEOUT 150)
    set_tests_properties(integration PROPERTIES SKIP_RETURN_CODE 77)
endif()
